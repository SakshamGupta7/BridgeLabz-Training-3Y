Q1. greetUser(name, callback)  -->  file: q1.js
=====================================================================

// Callback function: runs AFTER the greeting is printed
function showEndMessage() {
  console.log("Welcome to the course!");
}

// Main function: receives a name and a callback function
function greetUser(name, callback) {
  console.log("Hello " + name);   // Step 1: greeting is printed first
  callback();                      // Step 2: callback is executed afterwards
}

// Demonstration of callback flow:
// 1. greetUser is called  -> prints "Hello Saksham"
// 2. greetUser calls the function passed in -> prints "Welcome to the course!"
// Note: we pass showEndMessage WITHOUT () so it is not run immediately.
greetUser("Saksham", showEndMessage);

/* Expected output:
Hello Saksham
Welcome to the course!
*/

Q2. applyOperation(numbers, operation)  -->  file: q2.js
=====================================================================

// Takes an array and a callback; applies the callback to every element
// and returns a NEW array with the results.
function applyOperation(numbers, operation) {
  const result = [];
  for (let i = 0; i < numbers.length; i++) {
    result.push(operation(numbers[i])); // call the callback on each number
  }
  return result;
}

const numbers = [1, 2, 3, 4];

// Callback 1: double a number
const double = (num) => num * 2;

// Callback 2: square a number
const square = (num) => num * num;

console.log("Doubled:", applyOperation(numbers, double)); // [2, 4, 6, 8]
console.log("Squared:", applyOperation(numbers, square)); // [1, 4, 9, 16]

/* Expected output:
Doubled: [ 2, 4, 6, 8 ]
Squared: [ 1, 4, 9, 16 ]
*/

Q3. Arrow function vs normal function (this)  -->  file: q3.js
=====================================================================

// ---- Version 1: method written as an ARROW function ----
const user = {
  name: "Saksham",
  showName: () => {
    // Arrow functions do NOT have their own "this".
    // They use the "this" of the surrounding (outer) scope.
    console.log("Arrow function says:", this.name);
  },
};

user.showName(); // Output: Arrow function says: undefined

/*
WHY IS this.name UNDEFINED?
- An arrow function does not create its own "this". It inherits "this" from
  the scope where it was DEFINED, not from the object that calls it.
- The object literal { ... } is NOT a scope, so the arrow function's outer
  scope is the global/module scope.
- In Node.js (a module), "this" at the top level is an empty object
  (module.exports = {}), which has no "name" property -> undefined.
- In a browser, "this" is the window object (window.name is usually "").
- So "this" does not point to the user object, and this.name is undefined.
*/

// ---- Version 2 (FIX): method written as a NORMAL function ----
const userFixed = {
  name: "Saksham",
  showName: function () {
    // A normal function gets "this" from HOW it is called.
    // userFixed.showName() -> "this" is userFixed.
    console.log("Normal function says:", this.name);
  },
};

userFixed.showName(); // Output: Normal function says: Saksham

/* Expected output:
Arrow function says: undefined
Normal function says: Saksham
*/

Q4. Car constructor + prototype method  -->  file: q4.js
=====================================================================

// Constructor function: each car gets its own brand and model
function Car(brand, model) {
  this.brand = brand;
  this.model = model;
}

// Method added to the PROTOTYPE: stored only once in memory and
// shared by every Car object.
Car.prototype.getDetails = function () {
  console.log("Brand: " + this.brand + ", Model: " + this.model);
};

// Create two car objects
const car1 = new Car("Toyota", "Fortuner");
const car2 = new Car("Honda", "City");

car1.getDetails(); // Brand: Toyota, Model: Fortuner
car2.getDetails(); // Brand: Honda, Model: City

// Proof that the method is SHARED via the prototype (same function object):
console.log("Same method shared?", car1.getDetails === car2.getDetails); // true
console.log("getDetails is own property of car1?",
  car1.hasOwnProperty("getDetails")); // false (it lives on Car.prototype)

/* Expected output:
Brand: Toyota, Model: Fortuner
Brand: Honda, Model: City
Same method shared? true
getDetails is own property of car1? false
*/

Q5. Person -> Student inheritance (prototypes)  -->  file: q5.js
=====================================================================

// Parent constructor
function Person(name) {
  this.name = name;
}

// Parent method on prototype
Person.prototype.printName = function () {
  console.log("Name: " + this.name);
};

// Child constructor
function Student(name, branch) {
  Person.call(this, name); // inherit properties: run Person with Student's "this"
  this.branch = branch;    // property specific to Student
}

// Link the prototype chain: Student.prototype -> Person.prototype
Student.prototype = Object.create(Person.prototype);
// Restore the constructor reference (it gets overwritten by the line above)
Student.prototype.constructor = Student;

// Child method on prototype
Student.prototype.printBranch = function () {
  console.log("Branch: " + this.branch);
};

// Demonstration
const student1 = new Student("Saksham", "Computer Science");
student1.printName();    // inherited from Person.prototype
student1.printBranch();  // defined on Student.prototype

// Prototype chain proof:
console.log(student1 instanceof Student);                        // true
console.log(student1 instanceof Person);                         // true
console.log(Object.getPrototypeOf(student1) === Student.prototype); // true
console.log(Object.getPrototypeOf(Student.prototype) === Person.prototype); // true
// Chain: student1 -> Student.prototype -> Person.prototype -> Object.prototype -> null

/* Expected output:
Name: Saksham
Branch: Computer Science
true
true
true
true
*/

Q6. Prototype chain: Person -> Faculty -> Professor  -->  file: q6.js
=====================================================================

// ---- Level 1: Person ----
function Person(name) {
  this.name = name;
}
Person.prototype.introduce = function () {
  console.log("Hi, I am " + this.name);
};

// ---- Level 2: Faculty (inherits Person) ----
function Faculty(name, department) {
  Person.call(this, name);
  this.department = department;
}
Faculty.prototype = Object.create(Person.prototype);
Faculty.prototype.constructor = Faculty;
Faculty.prototype.showDepartment = function () {
  console.log(this.name + " works in the " + this.department + " department");
};

// ---- Level 3: Professor (inherits Faculty) ----
function Professor(name, department, researchArea) {
  Faculty.call(this, name, department);
  this.researchArea = researchArea;
}
Professor.prototype = Object.create(Faculty.prototype);
Professor.prototype.constructor = Professor;
Professor.prototype.showResearch = function () {
  console.log("Research area of " + this.name + ": " + this.researchArea);
};

// ---- Demonstration ----
const prof = new Professor("Dr. Sharma", "Computer Science", "Network Security");

prof.introduce();       // from Person.prototype    (2 levels up)
prof.showDepartment();  // from Faculty.prototype   (1 level up)
prof.showResearch();    // from Professor.prototype (own level)

console.log(prof instanceof Professor); // true
console.log(prof instanceof Faculty);   // true
console.log(prof instanceof Person);    // true
// Chain: prof -> Professor.prototype -> Faculty.prototype
//        -> Person.prototype -> Object.prototype -> null

/* Expected output:
Hi, I am Dr. Sharma
Dr. Sharma works in the Computer Science department
Research area of Dr. Sharma: Network Security
true
true
true
*/

Q7. makeMultiplier (closure)  -->  file: q7.js
=====================================================================

// Outer function takes a multiplier and RETURNS an inner function.
function makeMultiplier(multiplier) {
  // The inner function "remembers" the variable "multiplier"
  return function (number) {
    return number * multiplier;
  };
}

const double = makeMultiplier(2);
const triple = makeMultiplier(3);

console.log(triple(5)); // 15
console.log(double(5)); // 10

/*
HOW THE CLOSURE WORKS:
- When makeMultiplier(3) runs, it creates a local variable multiplier = 3
  and returns the inner function.
- Normally, local variables disappear after a function finishes. But the inner
  function still REFERENCES "multiplier", so JavaScript keeps that variable
  alive in memory.
- This combination of a function + the variables from the scope where it was
  created is called a CLOSURE.
- triple remembers multiplier = 3, and double remembers multiplier = 2.
  Each call to makeMultiplier creates a separate, private closure.
*/

/* Expected output:
15
10
*/

Q8. Custom Array.prototype.myMap()  -->  file: q8.js
=====================================================================

// Adds myMap to ALL arrays. It behaves like the built-in map():
// calls the callback for each element and returns a NEW array.
Array.prototype.myMap = function (callback) {
  const result = [];
  // "this" is the array on which myMap was called
  for (let i = 0; i < this.length; i++) {
    // Like built-in map, pass (element, index, array) to the callback
    result.push(callback(this[i], i, this));
  }
  return result;
};

// Demonstration
console.log([1, 2, 3].myMap((num) => num * 2));          // [2, 4, 6]
console.log(["a", "b"].myMap((ch, i) => ch + i));         // ["a0", "b1"]

// Original array is not changed (just like map)
const original = [10, 20, 30];
const mapped = original.myMap((n) => n / 10);
console.log(original); // [10, 20, 30]
console.log(mapped);   // [1, 2, 3]

/* Expected output:
[ 2, 4, 6 ]
[ 'a0', 'b1' ]
[ 10, 20, 30 ]
[ 1, 2, 3 ]
*/

Q9. Person -> Student using ES6 classes  -->  file: q9.js
=====================================================================

// ---------- ES6 CLASS VERSION ----------
class PersonClass {
  constructor(name) {
    this.name = name;
  }
  printName() {
    console.log("Name: " + this.name);
  }
}

class StudentClass extends PersonClass {
  constructor(name, branch) {
    super(name);          // calls the parent constructor (must come first)
    this.branch = branch;
  }
  printBranch() {
    console.log("Branch: " + this.branch);
  }
}

// ---------- PROTOTYPE VERSION (from Q5) ----------
function Person(name) {
  this.name = name;
}
Person.prototype.printName = function () {
  console.log("Name: " + this.name);
};

function Student(name, branch) {
  Person.call(this, name);
  this.branch = branch;
}
Student.prototype = Object.create(Person.prototype);
Student.prototype.constructor = Student;
Student.prototype.printBranch = function () {
  console.log("Branch: " + this.branch);
};

// ---------- DEMONSTRATION: both behave the same ----------
const classStudent = new StudentClass("Saksham", "Computer Science");
const protoStudent = new Student("Saksham", "Computer Science");

console.log("--- Class version ---");
classStudent.printName();
classStudent.printBranch();

console.log("--- Prototype version ---");
protoStudent.printName();
protoStudent.printBranch();

// Prove both have the same prototype-chain structure
console.log("--- Comparison ---");
console.log(classStudent instanceof PersonClass);  // true
console.log(protoStudent instanceof Person);       // true
console.log(
  Object.getPrototypeOf(StudentClass.prototype) === PersonClass.prototype
); // true
console.log(
  Object.getPrototypeOf(Student.prototype) === Person.prototype
); // true
// Methods live on the prototype in BOTH versions (not on the object itself)
console.log(classStudent.hasOwnProperty("printBranch")); // false
console.log(protoStudent.hasOwnProperty("printBranch")); // false

/*
CONCLUSION:
ES6 classes are "syntactic sugar" over prototypes. extends sets up the same
prototype link as Object.create(...), and super() does the same job as
Parent.call(this, ...). Under the hood, both versions behave identically.
*/

/* Expected output:
--- Class version ---
Name: Saksham
Branch: Computer Science
--- Prototype version ---
Name: Saksham
Branch: Computer Science
--- Comparison ---
true
true
true
true
false
false
*/
